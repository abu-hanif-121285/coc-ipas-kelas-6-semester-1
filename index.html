2. STRUKTUR GOOGLE SPREADSHEET (7 SHEET OTOMATIS)

Buat Google Spreadsheet baru bernama DB_IPAS_CHAMPIONS_2026. Script di bawah akan otomatis membuat 7 sheet beserta header-nya:

1. BANK_SOAL
2. PENGATURAN
3. DATA_PEMAIN
4. HASIL_PERTANDINGAN
5. RIWAYAT_JAWABAN
6. LEADERBOARD
7. RUBRIK_URAIAN

3. SOURCE CODE GOOGLE APPS SCRIPT (Code.gs)

Di Google Spreadsheet, klik Ekstensi → Apps Script, hapus kode lama, tempel kode berikut, lalu klik Deploy → Deployment Baru → Aplikasi Web (Akses: Siapa saja / Anyone).

// ============================================================================
// GOOGLE APPS SCRIPT BACKEND: CLASH OF CHAMPIONS - IPAS CHAMPIONS (KELAS 6 SD)
// Created by Wiyanto Abu Hanif
// ============================================================================

const SHEET_NAMES = [
  'BANK_SOAL',
  'PENGATURAN',
  'DATA_PEMAIN',
  'HASIL_PERTANDINGAN',
  'RIWAYAT_JAWABAN',
  'LEADERBOARD',
  'RUBRIK_URAIAN'
];

function setupSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const headers = {
    BANK_SOAL: ['ID', 'BABAK', 'NOMOR', 'MATERI', 'JENIS', 'PERTANYAAN', 'A', 'B', 'C', 'D', 'KUNCI', 'PEMBAHASAN', 'LEVEL', 'BOBOT', 'RUBRIK_URAIAN', 'JAWABAN_IDEAL'],
    PENGATURAN: ['KUNCI_PENGATURAN', 'NILAI', 'DIPERBARUI'],
    DATA_PEMAIN: ['TIMESTAMP', 'ROOM_CODE', 'NAMA', 'AVATAR', 'STATUS'],
    HASIL_PERTANDINGAN: ['TIMESTAMP', 'MATCH_ID', 'NAMA', 'SKOR_PG', 'SKOR_ISIAN', 'SKOR_URAIAN', 'NILAI_ASAS', 'POIN_KOMPETISI', 'BENAR', 'SALAH'],
    RIWAYAT_JAWABAN: ['TIMESTAMP', 'MATCH_ID', 'NAMA', 'NOMOR_SOAL', 'JENIS', 'JAWABAN_SISWA', 'STATUS_BENAR', 'SKOR_ASAS'],
    LEADERBOARD: ['PERINGKAT', 'NAMA', 'AVATAR', 'POIN_KOMPETISI', 'NILAI_ASAS', 'TANGGAL'],
    RUBRIK_URAIAN: ['ID_SOAL', 'NOMOR', 'PERTANYAAN', 'JAWABAN_IDEAL', 'RUBRIK_SKOR_0_4']
  };

  SHEET_NAMES.forEach(name => {
    let sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      sheet.appendRow(headers[name]);
      sheet.getRange(1, 1, 1, headers[name].length).setFontWeight('bold').setBackground('#0b1126').setFontColor('#00d2ff');
    }
  });
}

function doGet(e) {
  setupSheets();
  const action = (e && e.parameter && e.parameter.action) || 'ping';
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (action === 'ping') {
    return jsonResponse({ status: 'ok', message: 'GAS IPAS Champions Aktif!' });
  }

  // Ambil soal untuk pemain TANPA mengirim kunci jawaban & rubrik sebelum penilaian
  if (action === 'getQuestionsForPlayer') {
    const sheet = ss.getSheetByName('BANK_SOAL');
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return jsonResponse({ status: 'empty', questions: [] });
    const rows = data.slice(1).map(r => ({
      id: r[0], babak: Number(r[1]), nomor: Number(r[2]), materi: r[3],
      jenis: r[4], pertanyaan: r[5], a: r[6], b: r[7], c: r[8], d: r[9],
      level: r[12], bobot: Number(r[13])
    }));
    return jsonResponse({ status: 'ok', questions: rows });
  }

  // Ambil soal lengkap khusus Panel Guru (tervalidasi password)
  if (action === 'getQuestionsForTeacher') {
    if (e.parameter.pwd !== 'Inovatif') return jsonResponse({ status: 'error', message: 'Unauthorized' });
    const sheet = ss.getSheetByName('BANK_SOAL');
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return jsonResponse({ status: 'empty', questions: [] });
    const rows = data.slice(1).map(r => ({
      id: r[0], babak: Number(r[1]), nomor: Number(r[2]), materi: r[3],
      jenis: r[4], pertanyaan: r[5], a: r[6], b: r[7], c: r[8], d: r[9],
      kunci: r[10], pembahasan: r[11], level: r[12], bobot: Number(r[13]),
      rubrik: r[14], jawabanIdeal: r[15]
    }));
    return jsonResponse({ status: 'ok', questions: rows });
  }

  if (action === 'getLeaderboard') {
    const sheet = ss.getSheetByName('LEADERBOARD');
    const data = sheet.getDataRange().getValues().slice(1);
    return jsonResponse({ status: 'ok', leaderboard: data });
  }

  return jsonResponse({ status: 'error', message: 'Unknown action' });
}

function doPost(e) {
  setupSheets();
  const payload = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (payload.action === 'syncBankSoal') {
    if (payload.pwd !== 'Inovatif') return jsonResponse({ status: 'error', message: 'Unauthorized' });
    const sheet = ss.getSheetByName('BANK_SOAL');
    const rubrikSheet = ss.getSheetByName('RUBRIK_URAIAN');
    sheet.clearContents();
    rubrikSheet.clearContents();
    sheet.appendRow(['ID', 'BABAK', 'NOMOR', 'MATERI', 'JENIS', 'PERTANYAAN', 'A', 'B', 'C', 'D', 'KUNCI', 'PEMBAHASAN', 'LEVEL', 'BOBOT', 'RUBRIK_URAIAN', 'JAWABAN_IDEAL']);
    rubrikSheet.appendRow(['ID_SOAL', 'NOMOR', 'PERTANYAAN', 'JAWABAN_IDEAL', 'RUBRIK_SKOR_0_4']);

    payload.questions.forEach(q => {
      sheet.appendRow([q.id, q.babak, q.nomor, q.materi, q.jenis, q.pertanyaan, q.a || '', q.b || '', q.c || '', q.d || '', q.kunci, q.pembahasan, q.level, q.bobot, q.rubrik || '', q.jawabanIdeal || '']);
      if (q.jenis === 'URAIAN') {
        rubrikSheet.appendRow([q.id, q.nomor, q.pertanyaan, q.jawabanIdeal || q.kunci, q.rubrik || '']);
      }
    });
    return jsonResponse({ status: 'ok', message: '35 Soal berhasil disinkronkan ke Spreadsheet!' });
  }

  if (payload.action === 'saveMatchResult') {
    const resSheet = ss.getSheetByName('HASIL_PERTANDINGAN');
    const lbSheet = ss.getSheetByName('LEADERBOARD');
    const histSheet = ss.getSheetByName('RIWAYAT_JAWABAN');
    const now = new Date();

    payload.players.forEach((p, idx) => {
      resSheet.appendRow([now, payload.matchId, p.name, p.scorePG, p.scoreIsian, p.scoreUraian, p.academicScore, p.gamePoints, p.correctCount, p.wrongCount]);
      lbSheet.appendRow([idx + 1, p.name, p.avatar, p.gamePoints, p.academicScore, now]);
    });

    (payload.history || []).forEach(h => {
      histSheet.appendRow([now, payload.matchId, h.playerName, h.nomor, h.jenis, h.userAnswer, h.isCorrect ? 'BENAR' : 'SALAH', h.academicEarned]);
    });

    return jsonResponse({ status: 'ok', message: 'Hasil pertandingan tersimpan!' });
  }

  return jsonResponse({ status: 'error', message: 'Invalid POST action' });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
